async function loadStats() { try { const res = await fetch('/api/status'); const data = await res.json(); document.getElementById('statusText').textContent = `${data.service} | ${data.database}`; } catch (error) { document.getElementById('statusText').textContent = 'تعذر تحميل الحالة'; } } async function loadTournaments() { try { const res = await fetch('/api/tournaments'); const data = await res.json(); const list = document.getElementById('tournamentList'); if (!list) return; if (!data.length) { list.innerHTML = '
  • لا توجد بطولات حتى الآن.
    • '; return; } list.innerHTML = data.slice(0, 8).map(t => `
  • #${t.id} - ${t.name} - ${t.status}
    • `).join(''); } catch (error) { const list = document.getElementById('tournamentList'); if (list) list.innerHTML = '
  • تعذر تحميل البيانات.
    • '; } } window.addEventListener('DOMContentLoaded', () => { loadStats(); loadTournaments(); });
